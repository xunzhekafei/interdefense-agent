#
# Windows 上 ctranslate2 用 GPU 时需要 cuBLAS / cuDNN 的 DLL。
# Python 3.8+ 不再用 PATH 解析扩展模块的 DLL 依赖，pip 装的这两个包把 DLL 放在
# site-packages/nvidia/*/bin，需要 backend/whisper_asr.py 里显式 add_dll_directory 注册。
nvidia-cublas-cu12>=12,<13 ; sys_platform == "win32"
nvidia-cudnn-cu12>=9,<10 ; sys_platform == "win32"
